\subsection{Méthode}
\label{sec:Méthode}

\begin{frame}[c]{Méthode}
  \begin{columns}
    \column{.45\textwidth}
    \begin{itemize}
        \item Préciser le logiciel utilisé (ANSYS, Abaqus, ...)
        \item Préciser le type d'analyse (statique structurelle, modale, thermique, dynamique explicite, ...)
        \item Décrire la démarche complète lorsque la méthode ne se limite pas à une simple simulation (acquisition de données, identification des paramètres, essais, \textbf{validation} expérimentale, post-traitement, ...)
    \end{itemize}
    
    \column{.45\textwidth}
    \begin{itemize}
        \item Diagramme de flux (pour les méthodes complexes)
        \item draw.io ou TikZ, si possible
    \end{itemize}
  \end{columns}
\end{frame}
