\subsection{Validation des résultats}
\label{sec:Validation des résultats}

\begin{frame}[c]{Validation des résultats}


\begin{itemize}
    \item Comparaison avec un calcul analytique
    \item Comparaison avec des données expérimentales
    \item Comparaison avec la littérature ou des cas de référence
    \item Analyse de la plausibilité physique des résultats
    \item Quantification de l'écart observé
    \item \textit{Les résultats représentent-ils correctement le comportement réel ?}
\end{itemize}

\end{frame}
