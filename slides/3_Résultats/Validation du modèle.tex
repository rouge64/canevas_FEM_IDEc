\subsection{Validation du modèle}
\label{sec:Validation du modèle}

\begin{frame}[c]{Validation du modèle}

\begin{itemize}
    \item Vérification de l'équilibre des forces et des moments
    \item Réactions aux appuis cohérentes
    \item Vérification des unités et des ordres de grandeur
    \item Cas limites ou benchmarks simples
    \item Vérification des hypothèses de modélisation
    \item \textit{Le modèle numérique est-il correctement construit ?}
\end{itemize}

\end{frame}
